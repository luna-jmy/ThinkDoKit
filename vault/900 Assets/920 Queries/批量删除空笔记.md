# 批量删除空笔记

```dataviewjs
// 批量筛选并删除仅标题、无YAML、无内容的空笔记
(async () => {
    const app = this.app; // 兼容不同Dataview版本的作用域
    const vault = app.vault;
    
    // 定义移到回收站的函数（true = 系统回收站，false = Obsidian内部回收站）
    const moveToTrash = async (file) => {
        if (file) {
            await vault.trash(file, true);
            console.log(`已移到回收站：${file.path}`);
        }
    };

    // 核心筛选逻辑：无YAML + 无正文内容（仅标题）
    let emptyNotes = [];
    const allMdFiles = vault.getMarkdownFiles();
    
    for (const file of allMdFiles) {
        try {
            // 读取文件内容（异步读取避免卡顿）
            const content = await vault.cachedRead(file);
            // 筛选条件：
            // 1. 内容去除空白后为空 → 无正文
            // 2. 不以---开头 → 无YAML属性
            const contentTrimmed = content.trim();
            const hasNoContent = contentTrimmed === '';
            const hasNoYAML = !content.startsWith('---');
            
            if (hasNoContent && hasNoYAML) {
                emptyNotes.push(file);
            }
        } catch (e) {
            console.error(`读取文件失败：${file.path}`, e);
        }
    }

    // 渲染结果到笔记中
    dv.header(3, `找到 ${emptyNotes.length} 个符合条件的空笔记`);
    
    if (emptyNotes.length === 0) {
        dv.paragraph('✅ 没有找到仅标题、无YAML、无内容的空笔记');
    } else {
        // 1. 批量删除按钮
        const batchBtn = dv.el('button', '🗑️ 批量删除所有空笔记', { cls: 'mod-warning' });
        batchBtn.style.marginBottom = '10px';
        batchBtn.onclick = async () => {
            for (const file of emptyNotes) {
                await moveToTrash(file);
            }
            batchBtn.textContent = '✅ 已全部删除';
            batchBtn.disabled = true;
            // 刷新视图
            setTimeout(() => dv.reload(), 500);
        };
        
        // 2. 列出所有空笔记 + 单个删除按钮
        const tableData = emptyNotes.map(file => {
            // 单个删除按钮
            const singleBtn = dv.el('button', '删除', { cls: 'mod-danger' });
            singleBtn.onclick = async () => {
                await moveToTrash(file);
                singleBtn.textContent = '已删';
                singleBtn.disabled = true;
                setTimeout(() => dv.reload(), 500);
            };
            
            return [
                file.path, // 笔记路径
                singleBtn  // 操作按钮
            ];
        });
        
        // 渲染按钮和表格
        dv.el('div', batchBtn);
        dv.table(['笔记路径', '操作'], tableData);
    }
})();
```


## 优化带进度提示
```dataviewjs
// 高效版：带进度提示的批量删除空笔记（适配大量文件）
(async () => {
    const app = this.app;
    const vault = app.vault;
    let emptyNotes = [];
    let deletedCount = 0;

    // ========== 1. 先快速筛选空笔记（优化读取逻辑） ==========
    dv.header(3, '🔍 正在扫描空笔记...');
    const progressText = dv.el('p', '扫描进度：0%');
    const allFiles = vault.getMarkdownFiles();
    const totalFiles = allFiles.length;

    // 分批扫描，避免卡死
    for (let i = 0; i < totalFiles; i++) {
        const file = allFiles[i];
        // 每扫描20个文件更新一次进度（减少界面刷新消耗）
        if (i % 20 === 0) {
            progressText.textContent = `扫描进度：${Math.round((i/totalFiles)*100)}% (${i}/${totalFiles})`;
        }
        try {
            const content = await vault.cachedRead(file);
            // 筛选条件：无YAML + 无正文
            if (content.trim() === '' && !content.startsWith('---')) {
                emptyNotes.push(file);
            }
        } catch (e) {
            console.log(`跳过无法读取的文件：${file.path}`);
        }
    }

    // ========== 2. 渲染扫描结果 ==========
    progressText.remove(); // 移除扫描进度提示
    dv.header(3, `✅ 扫描完成！找到 ${emptyNotes.length} 个空笔记`);
    if (emptyNotes.length === 0) {
        dv.paragraph('无需删除任何笔记');
        return;
    }

    // ========== 3. 带进度的批量删除按钮 ==========
    const deleteBtn = dv.el('button', '🗑️ 开始批量删除（移到回收站）', {
        style: 'background: #dc2626; color: white; border: none; padding: 10px 20px; border-radius: 4px; cursor: pointer; font-size: 16px;'
    });
    const deleteProgress = dv.el('p', '删除进度：0/' + emptyNotes.length); // 实时进度提示
    dv.el('div', deleteBtn);
    dv.el('div', deleteProgress);

    deleteBtn.onclick = async () => {
        deleteBtn.disabled = true; // 禁用按钮，避免重复点击
        deleteBtn.textContent = '正在删除...';
        
        // 分批删除（每批50个，避免一次性处理卡死）
        const batchSize = 50;
        const totalBatches = Math.ceil(emptyNotes.length / batchSize);

        for (let batch = 0; batch < totalBatches; batch++) {
            const start = batch * batchSize;
            const end = Math.min((batch + 1) * batchSize, emptyNotes.length);
            const batchFiles = emptyNotes.slice(start, end);

            // 处理当前批次
            for (const file of batchFiles) {
                try {
                    await vault.trash(file, true); // true=系统回收站
                    deletedCount++;
                    // 更新进度
                    deleteProgress.textContent = `删除进度：${deletedCount}/${emptyNotes.length}（${Math.round((deletedCount/emptyNotes.length)*100)}%）`;
                } catch (e) {
                    console.error(`删除失败：${file.path} → ${e.message}`);
                    deleteProgress.textContent += `\n⚠️ 失败：${file.name}`;
                }
                // 短暂延迟，避免占用过多CPU
                await new Promise(resolve => setTimeout(resolve, 10));
            }
        }

        // ========== 4. 删除完成提示 ==========
        deleteBtn.textContent = '✅ 删除完成！';
        deleteProgress.textContent = `最终结果：成功删除 ${deletedCount} 个笔记，共 ${emptyNotes.length} 个待处理`;
        
        // 刷新界面
        setTimeout(() => {
            app.workspace.activeLeaf.reload();
        }, 1000);
    };
})();
```
