## 说明

闪卡格式：
- 填空闪卡： ==高亮文本==
- 单行闪卡： `;;` 翻转卡片 `;;;`
- 多行闪卡： `?` 翻转卡片 `??`

查找路径： `600 Zettelkasten` `300 Resources/330 Books`

---

## 闪卡笔记列表及统计数据
```dataviewjs
// 定义要查询的路径
let targetPaths = ['"600 Zettelkasten"', '"300 Resources/330 Books"'];

// 快速检测闪卡类型和数量
function detectFlashcards(content) {
    let cardCounts = {
        highlight: 0,
        singleLine: 0,
        multiLine: 0
    };
    
    // 检测填空闪卡
    let highlightMatches = content.match(/==.+?==/g);
    if (highlightMatches) {
        cardCounts.highlight = highlightMatches.length;
    }
    
    // 检测单行闪卡
    let lines = content.split('\n');
    lines.forEach(line => {
        if (line.includes(';;')) {
            cardCounts.singleLine++;
        }
    });
    
    // 检测多行闪卡
    lines.forEach(line => {
        let trimmed = line.trim();
        if (trimmed === '?' || trimmed === '??') {
            cardCounts.multiLine++;
        }
    });
    
    return cardCounts;
}

let grouped = {};
let totalStats = {
    files: 0,
    highlight: 0,
    singleLine: 0,
    multiLine: 0
};

// 遍历每个目标路径
for (let path of targetPaths) {
    let pages = dv.pages(path);
    
    for (let page of pages) {
        let content = await dv.io.load(page.file.path);
        if (content) {
            let cardCounts = detectFlashcards(content);
            let totalCards = cardCounts.highlight + cardCounts.singleLine + cardCounts.multiLine;
            
            // 如果找到闪卡，添加到结果中
            if (totalCards > 0) {
                let folder = page.file.folder || "根目录";
                if (!grouped[folder]) {
                    grouped[folder] = [];
                }
                
                let cardTypes = [];
                if (cardCounts.highlight > 0) cardTypes.push(`填空闪卡(${cardCounts.highlight})`);
                if (cardCounts.singleLine > 0) cardTypes.push(`单行闪卡(${cardCounts.singleLine})`);
                if (cardCounts.multiLine > 0) cardTypes.push(`多行闪卡(${cardCounts.multiLine})`);
                
                grouped[folder].push({
                    link: page.file.link,
                    totalCards: totalCards,
                    cardTypes: cardTypes,
                    counts: cardCounts
                });
                
                // 更新总统计
                totalStats.files++;
                totalStats.highlight += cardCounts.highlight;
                totalStats.singleLine += cardCounts.singleLine;
                totalStats.multiLine += cardCounts.multiLine;
            }
        }
    }
}

// 按文件夹显示结果
for (let folder in grouped) {
    dv.header(3, `📁 ${folder}`);
    
    // 创建包含统计信息的列表
    let listItems = grouped[folder].map(item => 
        `${item.link} - ${item.cardTypes.join(", ")} (共${item.totalCards}张)`
    );
    
    dv.list(listItems);
    
    // 显示文件夹统计
    let folderStats = grouped[folder].reduce((acc, item) => {
        acc.files++;
        acc.highlight += item.counts.highlight;
        acc.singleLine += item.counts.singleLine;
        acc.multiLine += item.counts.multiLine;
        return acc;
    }, { files: 0, highlight: 0, singleLine: 0, multiLine: 0 });
    
    let folderTotal = folderStats.highlight + folderStats.singleLine + folderStats.multiLine;
    dv.paragraph(`📊 **${folder}**: ${folderStats.files} 个文件，共 ${folderTotal} 张闪卡`);
    dv.paragraph("");
}

// 显示总体统计信息
if (Object.keys(grouped).length === 0) {
    dv.paragraph("❌ 未找到包含闪卡格式的笔记。");
} else {
    let grandTotal = totalStats.highlight + totalStats.singleLine + totalStats.multiLine;
    
    dv.header(3, "📈 总体统计");
    dv.paragraph(`🗂️ **总文件数**: ${totalStats.files} 个`);
    dv.paragraph(`📚 **总闪卡数**: ${grandTotal} 张`);
    
    if (totalStats.highlight > 0) {
        dv.paragraph(`　　• 填空闪卡: ${totalStats.highlight} 张`);
    }
    if (totalStats.singleLine > 0) {
        dv.paragraph(`　　• 单行闪卡: ${totalStats.singleLine} 张`);
    }
    if (totalStats.multiLine > 0) {
        dv.paragraph(`　　• 多行闪卡: ${totalStats.multiLine} 张`);
    }
    
    // 计算平均值
    let avgCards = (grandTotal / totalStats.files).toFixed(1);
    dv.paragraph(`📊 **平均每个文件**: ${avgCards} 张闪卡`);
}
```
